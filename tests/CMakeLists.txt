function(luma_add_test target source)
    add_executable(${target} ${source})
    target_include_directories(${target} PRIVATE ${PROJECT_SOURCE_DIR}/tests)
    target_link_libraries(${target} PRIVATE ${ARGN} luma_compile_options)
    add_test(NAME ${target} COMMAND ${target})
endfunction()

luma_add_test(test_types core/test_types.cpp luma::core)
luma_add_test(test_ring_buffer core/test_ring_buffer.cpp luma::core)
luma_add_test(test_inproc_event_bus core/test_inproc_event_bus.cpp luma::core)
luma_add_test(test_safety_guard safety/test_safety_guard.cpp luma::safety)
luma_add_test(test_nexarm_protocol execution/test_nexarm_protocol.cpp luma::execution)
